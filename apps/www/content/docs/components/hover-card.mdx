## Usage

```tsx
import { HoverCard, HoverCardTrigger, HoverCardContent } from "@/registry/wuhan/ui/hover-card"
```

```tsx
<HoverCard>
  <HoverCardTrigger asChild>
    <Button>Hover me</Button>
  </HoverCardTrigger>
  <HoverCardContent>Content</HoverCardContent>
</HoverCard>
```
