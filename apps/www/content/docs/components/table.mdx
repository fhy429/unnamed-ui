## Usage

```tsx
import { Table, TableHeader, TableBody, TableRow, TableHead, TableCell, TableFooter } from "@/registry/wuhan/ui/table"
```

```tsx
<Table>
  <TableHeader>
    <TableRow>
      <TableHead>Name</TableHead>
      <TableHead>Email</TableHead>
    </TableRow>
  </TableHeader>
  <TableBody>
    <TableRow>
      <TableCell>John Doe</TableCell>
      <TableCell>john@example.com</TableCell>
    </TableRow>
  </TableBody>
</Table>
```
