## Usage

```tsx
import { Item, ItemContent, ItemTitle, ItemDescription, ItemMedia, ItemActions } from "@/registry/wuhan/ui/item"
```

```tsx
<Item>
  <ItemMedia>Icon</ItemMedia>
  <ItemContent>
    <ItemTitle>Title</ItemTitle>
    <ItemDescription>Description</ItemDescription>
  </ItemContent>
  <ItemActions>
    <Button>Action</Button>
  </ItemActions>
</Item>
```
