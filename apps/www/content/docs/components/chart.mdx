## Usage

```tsx
import { ChartContainer, ChartTooltip, ChartTooltipContent } from "@/registry/wuhan/ui/chart"
import { BarChart, Bar, XAxis } from "recharts"
```

```tsx
<ChartContainer config={chartConfig}>
  <BarChart data={data}>
    <XAxis dataKey="name" />
    <ChartTooltip content={<ChartTooltipContent />} />
    <Bar dataKey="value" />
  </BarChart>
</ChartContainer>
```
