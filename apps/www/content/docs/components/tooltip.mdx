## Usage

```tsx
import { Tooltip, TooltipTrigger, TooltipContent } from "@/registry/wuhan/ui/tooltip"
```

```tsx
<Tooltip>
  <TooltipTrigger asChild>
    <Button>Hover me</Button>
  </TooltipTrigger>
  <TooltipContent>
    <p>Tooltip content</p>
  </TooltipContent>
</Tooltip>
```
