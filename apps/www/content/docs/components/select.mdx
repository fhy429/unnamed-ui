## Usage

```tsx
import { Select, SelectTrigger, SelectValue, SelectContent, SelectItem } from "@/registry/wuhan/ui/select"
```

```tsx
<Select>
  <SelectTrigger>
    <SelectValue placeholder="Select an option" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="option1">Option 1</SelectItem>
  </SelectContent>
</Select>
```
