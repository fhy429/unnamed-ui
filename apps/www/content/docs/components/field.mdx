## Usage

```tsx
import { Field, FieldLabel, FieldDescription, FieldGroup } from "@/registry/wuhan/ui/field"
```

```tsx
<FieldGroup>
  <Field>
    <FieldLabel htmlFor="email">Email</FieldLabel>
    <Input id="email" />
    <FieldDescription>Enter your email</FieldDescription>
  </Field>
</FieldGroup>
```
