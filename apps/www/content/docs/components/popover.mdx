## Usage

```tsx
import { Popover, PopoverTrigger, PopoverContent } from "@/registry/wuhan/ui/popover"
```

```tsx
<Popover>
  <PopoverTrigger asChild>
    <Button>Open</Button>
  </PopoverTrigger>
  <PopoverContent>Content</PopoverContent>
</Popover>
```
